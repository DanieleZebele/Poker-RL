# Ogni blocco e' incollabile da solo: comincia col cd.


# ── TRAINING ───────────────────────────────────────────────────────────────────
cd /mnt/shared_volume/POKER/Poker-RL
./run.sh start          # avvia il loop (gira per giorni, staccato per progetto)
./run.sh watch          # tabella dei worker, si ridisegna ogni 30s
./run.sh status         # una lettura sola
./run.sh stop           # finisce la generazione in corso (puo' volerci ~20 min)
./run.sh kill           # ferma tutto adesso, in qualunque stato
./run.sh kill-dry       # ... elenca cosa ucciderebbe, senza toccare niente


# ── WEBAPP (tutte le macchine insieme) ────────────────────────────────────────
cd /mnt/shared_volume/POKER/Poker-RL
OMP_NUM_THREADS=1 .venv/bin/poker-dashboard --host 0.0.0.0 --iterations 1000
# da qui:          http://127.0.0.1:8770
# dalle altre VM:  http://10.64.10.17:8770
# Clicca la riga di un worker per le sue curve. Ctrl-C per chiudere.
# --iterations deve combaciare con quello del loop in corso, o le barre sono
# scalate contro il bersaglio sbagliato.


# ── GUI ───────────────────────────────────────────────────────────────────────
cd /mnt/shared_volume/POKER/Poker-RL
DISPLAY=:1 .venv/bin/poker-gui


# ── RICALIBRARE ELO BENCHMARK ───────────────────────────

cd /mnt/shared_volume/POKER/Poker-RL
OMP_NUM_THREADS=1 PYTHONUNBUFFERED=1 .venv/bin/python -m pokerlab.rl.benchmark_arena \
    --games-per-model 20 --workers 20 \
    --tolerance 1 --window 20 --min-rounds 50 --max-rounds 200 \
    2>&1 | tee checkpoints/logs/arena-$(date +%Y%m%d-%H%M%S).log

# solo snapshot, senza rigiocare:
OMP_NUM_THREADS=1 .venv/bin/python -m pokerlab.rl.benchmark_arena --max-rounds 0

# annulla i rating (i file restano dove sono):
.venv/bin/python -m pokerlab.rl.benchmark_arena --restore \
    checkpoints/global/benchmark_arena_backups/ratings-<timestamp>.json


# ── AGGIORNARE L'ELO DI TUTTI I MODELLI (giro di fine training, a mano) ──────────────────────────

cd /mnt/shared_volume/POKER/Poker-RL
OMP_NUM_THREADS=1 PYTHONUNBUFFERED=1 .venv/bin/poker-elo \
    --rounds 100 --workers 20 --hands 1000 \
    2>&1 | tee checkpoints/logs/elo-$(date +%Y%m%d-%H%M%S).log

# ~95 min per giro con --hands 1000, e un giro siede ~55 modelli su ~9.200:
# per coprirli tutti servono ~170 giri. Le varianti:
#   --hands 200         giro da ~20 min, rating meno precisi ma molti piu' modelli
#   --coverage random   estrazione uniforme, come i giri automatici
#   --prune             permetti anche l'eliminazione, che CANCELLA i file


# ── QUANTE MANI SERVONO PER SAPERE CHI E' PIU' FORTE ──────────────────────────
cd /mnt/shared_volume/POKER/Poker-RL
OMP_NUM_THREADS=1 .venv/bin/python -m pokerlab.rl.duel_power \
    --model-a '#1' --model-b '#100' --streams 16 --hands 10000 --jobs 6 \
    --save /tmp/duel.json
# rianalizza senza rigiocare, ad altre lunghezze di partita:
.venv/bin/python -m pokerlab.rl.duel_power --load /tmp/duel.json --lengths 100,1000,50000
#   --mode field   un seggio a testa e 4 riempitivi, come un giro di popolazione
